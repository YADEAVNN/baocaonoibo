  <script>
    const token = '<?= token ?>';
    document.getElementById('ok').onclick = ()=>{
      const p1 = document.getElementById('p1').value.trim();
      const p2 = document.getElementById('p2').value.trim();
      const msg= document.getElementById('msg');
      if(!p1 || !p2) { msg.textContent='Vui lòng nhập đầy đủ.'; msg.style.color='#c00'; return; }
      if(p1!==p2){ msg.textContent='Mật khẩu nhập lại không khớp.'; msg.style.color='#c00'; return; }
      google.script.run.withSuccessHandler(res=>{
        msg.textContent = res.message || 'OK';
        msg.style.color='#0a0';
      }).withFailureHandler(err=>{
        msg.textContent = (err.message||err);
        msg.style.color='#c00';
      }).completePasswordReset(token, p1);
    };
  </script>
